    void
    ClemParticles::DoRegridding(){

        CreateParticleBeforeRegridding();
        amrex::Print() << "Finish creating particles before regridding" << std::endl;
        Redistribute();
        amrex::Print() << "Finish Redistribute" << std::endl;
        DefineOwnershipParticleMesh();
        amrex::Print() << "Finish DefineOwnershipParticleMesh" << std::endl;

        const int lev                   = 0;
        const auto& geom                = Geom(lev);
        const auto* dx                  = geom.CellSize();
        const auto* plo                 = geom.ProbLo();
        //     *  *  *          That why diviosn by N+1 lem element
        // -> --------- <-
        const amrex::Real dx_inner      = dx[0] / static_cast<amrex::Real>(NUM_LEM + 1);

        AMREX_ASSERT(m_cell_vector_updated);

        for(MyPartIter pti(*this, lev); pti.isValid(); ++pti){ 
            const int grid_id           = pti.index();
            const int tile_id           = pti.LocalTileIndex();
            auto& cell_fab              = m_cell_vectors[grid_id]; //Here the particles idex are stores
            const amrex::Box& bx        = pti.validbox();
            auto& particle_tile         = GetParticles(lev)[std::make_pair(grid_id, tile_id)];
            auto& particles             = particle_tile.GetArrayOfStructs();

            for(amrex::IntVect iv = bx.smallEnd(); iv <= bx.bigEnd();  bx.next(iv)){
                amrex::Print() << "Regridding cell: " << iv << std::endl;
                auto& plist         = cell_fab(iv);
                std::sort(plist.begin(), plist.end(),[&](int a, int b) {return particles[a].idata(IntData::idx)< particles[b].idata(IntData::idx);});
                
                amrex::Real total_mass  = 0.0;

                const int numberOfLem   = plist.size();
                //Verification if the index are well sortex 
                for (int n = 0; n < numberOfLem; ++n) {
                    const int pidx = plist[n];
                    const int idx  = particles[pidx].idata(IntData::idx);
                    AMREX_ASSERT_WITH_MESSAGE(idx == n, "Particle indices are not sorted correctly before regridding");
                }


                for(const int pidx : plist){
                    total_mass += particles[pidx].rdata(RealData::mass);
                }

                const amrex::Real new_dm     = total_mass/static_cast<amrex::Real>(NUM_LEM);

                amrex::Print()<< "Total mass: " << total_mass << " New dm: " << new_dm << " Current Number of lem: " << numberOfLem << std::endl;
                //This is auxiliar buffer: only sotres the remaining information
                //thai sid dm = is the new dm; 
                // i : index for the index before the regriddign
                // j : index for the reinitialized particles
                //mass_i > dm then auxiliar takes this  mass_i-dm,


                /*
                aux: This container is a buffer for data when a particle has more mass tahtn the required then stores the property and the remaining mass to use in teh nex iteration*/

                RegriddingUt::LemAccumulator aux; //that is particle-like object
                RegriddingUt::LemAccumulator acc; //This containes the mass*phi accumualtes taht later is computed using mass average

                //Regridding procedure
                int src = 0;
                int dst = 0;
                bool isActiveAux = false;
                acc.reset();
                aux.reset();

                
                while (dst < NUM_LEM) //Verify correctness
                {   
                    amrex::Print() << "dst: " << dst << " src: " << src << " isActiveAux: " << isActiveAux << std::endl;
                    ParticleType& psrc             = particles[plist[src]]; //to the cell i,j,k, the particles [31,21,45,18] in index {0,1,2,3} -> This rider always must be rpeserved
                    ParticleType& pdst             = particles[plist[dst]];

                    if (!isActiveAux){
                        amrex::Real m_src = psrc.rdata(RealData::mass);
                        amrex::Real m_need = new_dm - acc.mass;

                        if (m_src > m_need){
                            RegriddingUt::AccumulateDataIntoStructureGivenMassPortion(acc, psrc, m_need);
                            
                            //The particle reduces mass and volumes to be consistent 
                            psrc.rdata(RealData::mass)      -= m_need;
                            psrc.rdata(RealData::volume)    -= m_need / psrc.rdata(RealData::density);

                            RegriddingUt::StoreRemainningDataIntoAux(aux, psrc);
                            isActiveAux = true;
                            //This function set zero all the psdt and then wiht the accumualted data do a mass avergae
                            AMREX_ASSERT(std::abs(acc.mass - new_dm) < 1e-12);
                            RegriddingUt::UpdateLemFromAccumulator(acc, pdst, new_dm);
                            acc.reset();
                            dst++;
                        }
                        else{
                            RegriddingUt::AccumulateDataIntoStructure(acc, psrc);
                            psrc.rdata(RealData::mass) = 0.0;
                            src++;
                        }
                    }else{
                        amrex::Real m_aux = aux.mass;
                        amrex::Real m_need = new_dm - acc.mass;

                        if (m_aux > m_need){
                            RegriddingUt::AccumulateDataIntoStructureGivenMassPortion(acc, aux, m_need);
                            
                            //The particle reduces mass and volumes to be consistent 
                            aux.mass -= m_need;
                            aux.volume -= m_need / aux.density;
                            //This function set zero all the psdt and then wiht the accumualted data do a mass avergae
                            AMREX_ASSERT(std::abs(acc.mass - new_dm) < 1e-12);
                            RegriddingUt::UpdateLemFromAccumulator(acc, pdst, new_dm);
                            acc.reset();
                            dst++;
                        }
                        else{
                            RegriddingUt::AccumulateDataIntoStructure(acc, aux);
                            aux.reset();
                            isActiveAux = false;
                            src++;
                        }
                    }
                }
                amrex::Print() << "Regridding procedure done for cell: " << iv << std::endl;
                //verification of mass
                const int numberOfLemUpdated = plist.size();
                for(int lem = 0; lem < numberOfLemUpdated; lem++){
                    if(particles[plist[lem]].rdata(RealData::mass) - new_dm > 1e-12){
                        amrex::Print() << " Incorrect mass fill : "<< particles[plist[lem]].rdata(RealData::mass) << std::endl;
                    };

                    //LemParticles that are already empty need to be deleted
                    //Note that only until NUM_LEM are filled
                    if(!(lem< NUM_LEM)){
                        ParticleType& p = particles[plist[lem]];
                        p.id()          = p.id(); 
                        if(p.rdata(RealData::mass > 1e-12)){
                            amrex::Print() << " Existing mass : "<< particles[plist[lem]].rdata(RealData::mass) << "  lem: " << lem << std::endl;
                        }
                    }
                }


            }// box iteration
        }//pti iter
        amrex::Print() << "Regridding procedure done for all the cells " << std::endl;
        Redistribute();
    }//DoRegridding
